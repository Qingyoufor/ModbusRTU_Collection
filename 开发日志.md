### 步骤四：Program.cs 基础配置

```csharp
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.EntityFrameworkCore;
using Microsoft.IdentityModel.Tokens;
using Microsoft.OpenApi;
using Server.Infrastructure.Data;
using Server.Infrastructure.Mappings;
using System.Text;

var builder = WebApplication.CreateBuilder(args);

// ====== CORS ======
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowClient", policy =>
    {
        policy.WithOrigins("http://localhost:5000")
              .AllowAnyHeader()
              .AllowAnyMethod()
              .AllowCredentials();
    });
});

// ====== Controllers + Swagger ======
builder.Services.AddControllers();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen(options =>
{
    options.SwaggerDoc("v1", new OpenApiInfo { Title = "SmartFactory API", Version = "v1" });
    options.AddSecurityDefinition("Bearer", new OpenApiSecurityScheme
    {
        Description = "JWT Authorization header. Example: \"Bearer {token}\"",
        Name = "Authorization",
        In = ParameterLocation.Header,
        Type = SecuritySchemeType.ApiKey,
        Scheme = "Bearer"
    });
});

// ====== EF Core ======
builder.Services.AddDbContext<SmartFactoryDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));

// ====== AutoMapper ======
builder.Services.AddAutoMapper(cfg => cfg.AddMaps(typeof(MappingProfile).Assembly));

// ====== JWT 认证占位（Day2 完善）=====
var jwtSettings = builder.Configuration.GetSection("Jwt");
builder.Services.AddAuthentication(options =>
{
    options.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;
    options.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;
})
.AddJwtBearer(options =>
{
    options.TokenValidationParameters = new()
    {
        ValidateIssuerSigningKey = true,
        IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtSettings["Key"]!)),
        ValidateIssuer = true,
        ValidIssuer = jwtSettings["Issuer"],
        ValidateAudience = true,
        ValidAudience = jwtSettings["Audience"],
        ValidateLifetime = true,
        ClockSkew = TimeSpan.Zero,
    };
});

builder.Services.AddAuthorization();

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI(c => c.SwaggerEndpoint("/swagger/v1/swagger.json", "SmartFactory API v1"));
}

app.UseCors("AllowClient");
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();
app.Run();
```

>   学习心得  ：`builder.Services.AddXxx()` 是注册阶段，往 DI 容器里放服务；`builder.Build()` 是构建阶段，生成 `IServiceProvider`；`app.UseXxx()` 是使用阶段，按顺序组装中间件管道。

### 步骤八：SmartFactoryDbContext

```csharp
public class SmartFactoryDbContext : DbContext
{
    public SmartFactoryDbContext(DbContextOptions<SmartFactoryDbContext> options) : base(options) { }

    public DbSet<ProductionLine> productionLines => Set<ProductionLine>();
    public DbSet<Device> Devices => Set<Device>();
    public DbSet<DataPoint> DataPoints => Set<DataPoint>();
    public DbSet<User> Users => Set<User>();
    public DbSet<RefreshToken> RefreshTokens => Set<RefreshToken>();
    public DbSet<DeviceData> DeviceDatas => Set<DeviceData>();
    public DbSet<Alarm> Alarms => Set<Alarm>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // 未声明长度的 string 字段默认最大 256
        foreach (var entity in modelBuilder.Model.GetEntityTypes())
        {
            foreach (var prop in entity.GetProperties()
                .Where(p => p.ClrType == typeof(string)))
            {
                if (!prop.GetMaxLength().HasValue && !prop.IsPrimaryKey())
                    prop.SetMaxLength(256);
            }
        }

        // 级联删除限制（防止误删子数据）
        modelBuilder.Entity<Alarm>()
            .HasOne(a => a.Device).WithMany()
            .HasForeignKey(a => a.DeviceId).OnDelete(DeleteBehavior.Restrict);
        modelBuilder.Entity<Alarm>()
            .HasOne(a => a.DataPoint).WithMany()
            .HasForeignKey(a => a.DataPointId).OnDelete(DeleteBehavior.Restrict);
        modelBuilder.Entity<DeviceData>()
            .HasOne(d => d.Device).WithMany()
            .HasForeignKey(d => d.DeviceId).OnDelete(DeleteBehavior.Restrict);
        modelBuilder.Entity<DeviceData>()
            .HasOne(d => d.DataPoint).WithMany()
            .HasForeignKey(d => d.DataPointId).OnDelete(DeleteBehavior.Restrict);

        // 复合索引
        modelBuilder.Entity<DeviceData>()
            .HasIndex(d => new { d.DeviceId, d.DataPointId, d.RecordedAt })
            .HasDatabaseName("IX_DeviceDatas_Device_Point_Time");
        modelBuilder.Entity<Alarm>()
            .HasIndex(a => new { a.DeviceId, a.Status, a.OccurredAt })
            .HasDatabaseName("IX_Alarms_Device_Status_Time");

        // 种子数据
        var seedTime = new DateTime(2025, 1, 1, 0, 0, 0, DateTimeKind.Utc);
        modelBuilder.Entity<ProductionLine>().HasData(
            new ProductionLine { Id = 1, Name = "一号生产线", Description = "CNC加工中心", CreatedAt = seedTime },
            new ProductionLine { Id = 2, Name = "二号生产线", Description = "注塑机", CreatedAt = seedTime },
            new ProductionLine { Id = 3, Name = "三号生产线", Description = "环境监测", CreatedAt = seedTime }
        );
        modelBuilder.Entity<User>().HasData(
            new User
            {
                Id = 1,
                Username = "admin",
                PasswordHash = "$2a$11$fgGNAdEqb6qn8FOOJt4E2OgRzmqqdz/lNMZiInsJ1oYp5IY0Umeqe",
                Role = UserRole.Admin,
                IsActive = true,
                CreatedAt = seedTime
```

### 步骤二：Repository 泛型基类

```csharp
// IRepositoryBase.cs
public interface IRepositoryBase<TEntity> where TEntity : class
{
    Task<TEntity?> GetByIdAsync(int id);
    Task<List<TEntity>> GetAllAsync();
    Task<PagedResult<TEntity>> GetPagedAsync(int page, int pageSize);
    Task<TEntity> AddAsync(TEntity entity);
    void Update(TEntity entity);
    Task DeleteAsync(int id);
    Task SaveChangedAsync();
}

// RepositoryBase.cs
public class RepositoryBase<TEntity> : IRepositoryBase<TEntity> where TEntity : class
{
    protected readonly SmartFactoryDbContext _context;
    protected readonly DbSet<TEntity> _dbSet;

    public RepositoryBase(SmartFactoryDbContext context)
    {
        _context = context;
        _dbSet = context.Set<TEntity>();
    }

    public async Task<TEntity?> GetByIdAsync(int id) => await _dbSet.FindAsync(id);
    public async Task<List<TEntity>> GetAllAsync() => await _dbSet.ToListAsync();

    public async Task<PagedResult<TEntity>> GetPagedAsync(int page, int pageSize)
    {
        var query = _dbSet.AsQueryable();
        var totalCount = await query.CountAsync();
        var items = await query.Skip((page - 1) * pageSize).Take(pageSize).ToListAsync();
        return new PagedResult<TEntity>
        {
            Items = items, TotalCount = totalCount, Page = page, PageSize = pageSize
        };
    }

    public async Task<TEntity> AddAsync(TEntity entity)
    {
        await _dbSet.AddAsync(entity);
        return entity;
    }

    public void Update(TEntity entity) => _dbSet.Update(entity);

    public async Task DeleteAsync(int id)
    {
        var entity = await GetByIdAsync(id);
        if (entity != null) _dbSet.Remove(entity);
    }

    public async Task SaveChangedAsync() => await _context.SaveChangesAsync();
}
```

>   Repository 层职责  ：封装对 Entity 的 CRUD 操作，返回 Entity 类型。Service 层通过 Repository 操作 Entity，但不直接返回 Entity（而是映射为 DTO）。Controller 层不接收 / 返回 Entity。

### 步骤四：JwtHelper（纯静态工具类）

```csharp
public class JwtHelper
{
    // 生成 AccessToken（有效期 2 小时，无状态）
    public static string GenerateAccessToken(User user, IConfiguration jwtConfig)
    {
        var claims = new Claim[]
        {
            new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new Claim(ClaimTypes.Name, user.Username),
            new Claim(ClaimTypes.Role, user.Role.ToString()),
            new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
        };

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtConfig["Key"]!));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: jwtConfig["Issuer"],
            audience: jwtConfig["Audience"],
            claims: claims,
            expires: DateTime.UtcNow.AddMinutes(int.Parse(jwtConfig["ExpireMinutes"]!)),
            signingCredentials: creds);

        return new JwtSecurityTokenHandler().WriteToken(token);
    }

    // 生成 RefreshToken（随机字符串，有状态，存数据库）
    public static string GenerateRefreshValueToken()
        => Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));

    // 从过期 AccessToken 解析 Claims（只验证签名，不验证过期时间）
    public static ClaimsPrincipal GetClaimsPrincipalFromExpiredToken(string accessToken, string secretKey)
    {
        var parameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(secretKey)),
            ValidateIssuer = false,
            ValidateAudience = false,
            ValidateLifetime = false,
            ClockSkew = TimeSpan.Zero,
        };

        var handler = new JwtSecurityTokenHandler();
        var principal = handler.ValidateToken(accessToken, parameters, out _);
        return principal;
    }
}
```

>   为什么 JwtHelper 是静态类？   它不依赖 DI 容器，Service 层读取配置后传入参数即可。这样解耦了 JWT 逻辑和框架依赖。

### 步骤五：IAuthService / AuthService

```csharp
// IAuthService.cs
public interface IAuthService
{
    Task<ApiResult<LoginResponse>> LoginAsync(SignInRequest request);
    Task<ApiResult<LoginResponse>> RefreshTokenAsync(RefreshTokenRequest request);
    Task RevokeAllTokensAsync(int userId);
}

// AuthService.cs
public class AuthService : IAuthService
{
    private readonly IUserRepository _userRepo;
    private readonly IRefreshTokenRepository _tokenRepo;
    private readonly IMapper _mapper;
    private readonly IConfiguration _jwtConfig;

    public AuthService(IUserRepository userRepo, IRefreshTokenRepository tokenRepo,
        IMapper mapper, IConfiguration jwtConfig)
    {
        _userRepo = userRepo; _tokenRepo = tokenRepo; _mapper = mapper; _jwtConfig = jwtConfig;
    }

    public async Task<ApiResult<LoginResponse>> LoginAsync(SignInRequest request)
    {
        // 1. 校验账号密码
        var user = await _userRepo.GetByUsernameAsync(request.Username);
        if (user == null || !BCrypt.Net.BCrypt.Verify(request.Password, user.PasswordHash))
            return ApiResult<LoginResponse>.Fail(401, "账号或密码错误");

        // 2. 更新最后登录时间（EF 自动追踪）
        user.LastLoginAt = DateTime.UtcNow;
        _userRepo.Update(user);
        await _userRepo.SaveChangedAsync();

        // 3. 签发双 Token
        var jwtSection = _jwtConfig.GetSection("Jwt");
        var accessToken = JwtHelper.GenerateAccessToken(user, jwtSection);
        var refreshValue = JwtHelper.GenerateRefreshValueToken();

        // 4. 保存 RefreshToken
        await _tokenRepo.AddAsync(new RefreshToken
        {
            UserId = user.Id,
            Token = refreshValue,
            ExpiresAt = DateTime.UtcNow.AddDays(double.Parse(jwtSection["RefreshExpireDays"]!)),
            CreatedAt = DateTime.UtcNow
        });
        await _tokenRepo.SaveChangedAsync();

        return ApiResult<LoginResponse>.Success(new LoginResponse
        {
            AccseeToken = accessToken,
            RefreshToken = refreshValue,
            ExpiresAt = DateTime.UtcNow.AddMinutes(int.Parse(jwtSection["ExpireMinutes"]!)),
            User = _mapper.Map<UserDto>(user)
        });
    }

    public async Task<ApiResult<LoginResponse>> RefreshTokenAsync(RefreshTokenRequest request)
    {
        var jwtSection = _jwtConfig.GetSection("Jwt");

        // 1. 从过期 AccessToken 获取 UserId
        var principal = JwtHelper.GetClaimsPrincipalFromExpiredToken(request.AccessToken, jwtSection["Key"]!);
        var userIdStr = principal.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        if (userIdStr == null)
            return ApiResult<LoginResponse>.Fail(401, "无效的访问令牌");

        // 2. 验证 RefreshToken（未过期 + 未撤销）
        var storedToken = await _tokenRepo.GetByTokenValueAsync(request.RefreshToken);
        if (storedToken == null || storedToken.UserId.ToString() != userIdStr)
            return ApiResult<LoginResponse>.Fail(401, "无效的刷新令牌");

        // 3. 撤销旧 RefreshToken（一次性使用）
        storedToken.RevokedAt = DateTime.UtcNow;

        // 4. 签发新双 Token
        var user = await _userRepo.GetByIdAsync(int.Parse(userIdStr));
        if (user == null || !user.IsActive)
            return ApiResult<LoginResponse>.Fail(401, "用户不存在或已禁用");

        var newAccess = JwtHelper.GenerateAccessToken(user, jwtSection);
        var newRefresh = JwtHelper.GenerateRefreshValueToken();
        await _tokenRepo.AddAsync(new RefreshToken
        {
            UserId = user.Id, Token = newRefresh,
            ExpiresAt = DateTime.UtcNow.AddDays(double.Parse(jwtSection["RefreshExpireDays"]!)),
            CreatedAt = DateTime.UtcNow
        });
        await _tokenRepo.SaveChangedAsync();

        return ApiResult<LoginResponse>.Success(new LoginResponse
        {
            AccseeToken = newAccess, RefreshToken = newRefresh,
            ExpiresAt = DateTime.UtcNow.AddMinutes(int.Parse(jwtSection["ExpireMinutes"]!)),
            User = _mapper.Map<UserDto>(user)
        });
    }

    public async Task RevokeAllTokensAsync(int userId)
    {
        await _tokenRepo.RevokeAllUserTokensAsync(userId);
        await _tokenRepo.SaveChangedAsync();
    }
}
```

>   双令牌机制  ：
>
> - AccessToken（2 小时）：无状态，中间件自动验证，用于日常接口访问
> - RefreshToken（7 天滑动窗口）：有状态存数据库，用于刷新 AccessToken
> - RefreshToken 一次性使用：刷新后旧 Token 设 `RevokedAt` 作废，防止重放攻击

### 步骤八：ModbusConfigLoader（配置加载器）

```csharp
public class ModbusConfigLoader
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<ModbusConfigLoader> _logger;

    public ModbusConfigLoader(IServiceScopeFactory scopeFactory, ILogger<ModbusConfigLoader> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    public async Task<List<ModbusDeviceConfig>> LoadConfigsAsync()
    {
        using var scope = _scopeFactory.CreateScope();
        var deviceRepo = scope.ServiceProvider.GetRequiredService<IDeviceRepository>();
        var dataPointRepo = scope.ServiceProvider.GetRequiredService<IDataPointRepository>();

        var devices = await deviceRepo.GetAllAsync();
        var onlineDevices = devices.Where(d => d.Status != DeviceStatus.Offline).ToList();
        if (onlineDevices.Count == 0)
        {
            _logger.LogWarning("没有在线的设备");
            return new List<ModbusDeviceConfig>();
        }

        var configTasks = onlineDevices.Select(async device =>
        {
            var points = await dataPointRepo.GetByDeviceIdAsync(device.Id);
            return new ModbusDeviceConfig
            {
                DeviceId = device.Id,
                Name = device.Name,
                SlaveAddress = device.SlaveAddress,
                SlaveCOM = device.ComPort,
                DataPoints = points.Select(p => new DataPointConfig
                {
                    DataPointId = p.Id,
                    Name = p.Name,
                    RegisterAddress = ConvertToProtocolAddress(p.RegisterAddress, p.RegisterType),
                    RegisterType = p.RegisterType,
                    DataType = p.DataType,
                    ScaleFactor = p.ScaleFactor,
                    DecimalPlaces = p.DecimalPlaces,
                    MinValue = p.MinValue,
                    MaxValue = p.MaxValue
                }).OrderBy(p => p.RegisterAddress).ToList()
            };
        });

        var configs = (await Task.WhenAll(configTasks)).ToList();
        _logger.LogInformation("已加载 {DeviceCount} 个设备，共 {PointCount} 个测点",
            configs.Count, configs.Sum(d => d.DataPoints.Count));
        return configs;
    }

    // PLC 绝对地址 -> 协议地址（40001->0, 30001->0）
    private static int ConvertToProtocolAddress(int registerAddress, RegisterType type)
        => type == RegisterType.HoldingRegister
            ? registerAddress - 40001
            : registerAddress - 30001;
}
```

### 步骤九：ModbusPortManager（串口管理器）

```csharp
public class ModbusPortManager : IDisposable
{
    private readonly ILogger<ModbusPortManager> _logger;
    private readonly IConfiguration _config;
    private readonly Dictionary<string, SerialPort> _portPool = new();

    public ModbusPortManager(ILogger<ModbusPortManager> logger, IConfiguration config)
    {
        _logger = logger;
        _config = config;
    }

    public void InitializePorts(IEnumerable<string> slaveCOMs)
    {
        var mappings = BuildComMapping();
        foreach (var slaveCOM in slaveCOMs.Distinct())
        {
            if (!mappings.TryGetValue(slaveCOM, out var masterCOM))
            {
                _logger.LogWarning("从站端口 {SlavePort} 无对应的主站端口映射", slaveCOM);
                continue;
            }
            if (_portPool.ContainsKey(masterCOM))
            {
                _logger.LogDebug("主站端口 {MasterPort} 已打开，跳过", masterCOM);
                continue;
            }
            try
            {
                var port = CreateSerialPort(masterCOM);
                port.Open();
                _portPool[masterCOM] = port;
                _logger.LogInformation("主站端口 {MasterPort} 打开成功（对应从站 {SlavePort}）", masterCOM, slaveCOM);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "主站端口 {MasterPort} 打开失败", masterCOM);
            }
        }
    }

    public SerialPort? GetPort(string masterCOM)
        => _portPool.TryGetValue(masterCOM, out var port) && port.IsOpen ? port : null;

    public IReadOnlyDictionary<string, SerialPort> GetOpenPorts()
        => _portPool.Where(kv => kv.Value.IsOpen).ToDictionary(kv => kv.Key, kv => kv.Value);

    private Dictionary<string, string> BuildComMapping()
    {
        var mappings = _config.GetSection("Modbus:PortMappings")
            .Get<Dictionary<string, string>>() ?? new Dictionary<string, string>();
        return mappings.ToDictionary(m => m.Value, m => m.Key);
    }

    private SerialPort CreateSerialPort(string portName)
    {
        var section = _config.GetSection("Modbus:DefaultSerialPort");
        return new SerialPort(portName)
        {
            BaudRate = section.GetValue<int>("BaudRate"),
            DataBits = section.GetValue<int>("DataBits"),
            StopBits = Enum.Parse<StopBits>(section["StopBits"]!),
            Parity = Enum.Parse<Parity>(section["Parity"]!),
            ReadTimeout = section.GetValue<int>("ReadTimeout"),
            WriteTimeout = section.GetValue<int>("WriteTimeout")
        };
    }

    public void Dispose()
    {
        foreach (var port in _portPool.Values)
        {
            try { if (port.IsOpen) port.Close(); port.Dispose(); }
            catch { }
        }
        _portPool.Clear();
    }
}
```

>   端口映射  ：配置文件 `{ "COM1": "COM2" }` 表示主站打开 COM1 与从站 COM2 通信。`BuildComMapping()` 反转映射为 `{ "COM2": "COM1" }`，方便根据从站端口查找主站端口。

### 步骤十：ModbusDataReader（数据读取器）

```csharp
public class ModbusDataReader
{
    private readonly ILogger<ModbusDataReader> _logger;
    private readonly ModbusDataCache _cache;
    private readonly Dictionary<string, IModbusSerialMaster> _masterCache = new();

    public ModbusDataReader(ILogger<ModbusDataReader> logger, ModbusDataCache cache)
    {
        _logger = logger;
        _cache = cache;
    }

    public IModbusSerialMaster GetOrCreateMaster(SerialPort port)
    {
        if (_masterCache.TryGetValue(port.PortName, out var master))
            return master;

        var adapter = new SerialPortAdapter(port);
        var factory = new ModbusFactory();
        var transport = factory.CreateRtuTransport(adapter);
        master = factory.CreateMaster(transport);

        _masterCache[port.PortName] = master;
        _logger.LogDebug("创建 Modbus Master: {Port}", port.PortName);
        return master;
    }

    public async Task PollDeviceAsync(IModbusSerialMaster master, ModbusDeviceConfig device, CancellationToken token)
    {
        try
        {
            var groups = ModbusGroupingHelper.GroupContinousRegister(device.DataPoints);
            if (groups.Count == 0) return;

            foreach (var group in groups)
            {
                if (token.IsCancellationRequested) break;
                try
                {
                    await ReadGroupWithRetryAsync(master, device, group, token);
                }
                catch (Exception ex) when (ex is not OperationCanceledException)
                {
                    _logger.LogError(ex, "设备 {Device} 分组读取失败 [起始:{Start} 长度:{Length}]",
                        device.Name, group.StartAddress, group.GroupLength);
                }
            }
        }
        catch (Exception ex) when (ex is not OperationCanceledException)
        {
            _logger.LogError(ex, "采集设备 {Device} (从站:{Address}) 失败", device.Name, device.SlaveAddress);
        }
    }

    private async Task ReadGroupWithRetryAsync(IModbusSerialMaster master, ModbusDeviceConfig device,
        RegisterReadGroup group, CancellationToken token, int maxRetries = 2)
    {
        ushort[]? registers = null;
        Exception? lastException = null;

        for (int retry = 0; retry < maxRetries; retry++)
        {
            try
            {
                registers = await ReadRegistersAsync(master, device.SlaveAddress, group);
                break;
            }
            catch (Exception ex) when (ex is not OperationCanceledException)
            {
                lastException = ex;
                if (retry < maxRetries - 1)
                {
                    _logger.LogWarning("设备 {Device} 读取失败，第 {Retry} 次重试", device.Name, retry + 1);
                    await Task.Delay(50 * (retry + 1), token);
                }
            }
        }

        if (registers == null)
            throw new Exception($"重试 {maxRetries} 次后仍失败", lastException);

        UpdateCacheFromGroup(device, group, registers);
    }

    private async Task<ushort[]> ReadRegistersAsync(IModbusSerialMaster master, byte slaveAddress, RegisterReadGroup group)
    {
        return group.RegisterType == RegisterType.HoldingRegister
            ? await master.ReadHoldingRegistersAsync(slaveAddress, (ushort)group.StartAddress, (ushort)group.GroupLength)
            : await master.ReadInputRegistersAsync(slaveAddress, (ushort)group.StartAddress, (ushort)group.GroupLength);
    }

    private void UpdateCacheFromGroup(ModbusDeviceConfig device, RegisterReadGroup group, ushort[] registers)
    {
        foreach (var point in group.DataPoints)
        {
            try
            {
                int offset = point.RegisterAddress - group.StartAddress;
                int regCount = point.DataType == DataPointType.Int32 || point.DataType == DataPointType.Float ? 2 : 1;
                var pointRegisters = new ushort[regCount];
                Array.Copy(registers, offset, pointRegisters, 0, regCount);

                double value = ModbusDataParser.ParseRawValue(pointRegisters, point);
                if (!ModbusDataParser.IsInRange(value, point))
                    _logger.LogWarning("测点 {Point} 值 {Value} 超出范围 [{Min}, {Max}]",
                        point.Name, value, point.MinValue, point.MaxValue);

                _cache.UpdateDataPoint(device.DeviceId, point.DataPointId, value);
                _logger.LogDebug("设备 {Device} -> {Point} = {Value}", device.Name, point.Name, value);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "解析测点 {Point} 失败", point.Name);
            }
        }
    }

    public void ClearMasterCache()
    {
        foreach (var master in _masterCache.Values)
        { try { master.Dispose(); } catch { } }
        _masterCache.Clear();
    }
}
```

### 步骤十一：ModbusDataParser（数据解析器）

```csharp
public class ModbusDataParser
{
    public static double ParseRawValue(ushort[] registers, DataPointConfig config)
    {
        double rawValue = config.DataType switch
        {
            DataPointType.Int16 => (short)registers[0],
            DataPointType.UInt16 => registers[0],
            DataPointType.Int32 => (int)((uint)registers[0] << 16 | registers[1]),
            DataPointType.Float => ParseFloatBigEndian(registers),
            _ => registers[0]
        };

        double scaleValue = rawValue * config.ScaleFactor;
        return Math.Round(scaleValue, config.DecimalPlaces);
    }

    private static float ParseFloatBigEndian(ushort[] registers)
    {
        uint intValue = ((uint)registers[0] << 16) | registers[1];
        byte[] bytes = BitConverter.GetBytes(intValue);
        return BitConverter.ToSingle(bytes, 0);
    }

    public static bool IsInRange(double value, DataPointConfig config)
        => value >= config.MinValue && value <= config.MaxValue;
}
```

>   Float 解析关键点  ：`BitConverter` 是平台相关的（Windows 小端 / ARM 大端）。先通过 `((uint)registers[0] << 16) | registers[1]` 组合成平台无关的 `uint`，再用 `BitConverter.GetBytes(uint)` 生成当前平台正确字节序的数组，最后 `ToSingle` 解析。这避免了直接构造字节数组导致的字节序错误。

### 步骤十二：ModbusGroupingHelper（测点分组优化）

```csharp
public class ModbusGroupingHelper
{
    public static List<RegisterReadGroup> GroupContinousRegister(List<DataPointConfig> points)
    {
        if (points.Count == 0) return new List<RegisterReadGroup>();

        var sortedPoints = points.OrderBy(p => p.RegisterAddress).ToList();
        var groups = new List<RegisterReadGroup>();

        var groupStartAddr = sortedPoints[0].RegisterAddress;
        var groupPoints = new List<DataPointConfig> { sortedPoints[0] };
        var groupType = sortedPoints[0].RegisterType;
        int expectedNextAddr = groupStartAddr + GetRegisterLength(sortedPoints[0]);

        for (int i = 1; i < sortedPoints.Count; i++)
        {
            var point = sortedPoints[i];
            int pointLength = GetRegisterLength(point);

            if (point.RegisterType == groupType && point.RegisterAddress == expectedNextAddr)
            {
                groupPoints.Add(point);
                expectedNextAddr += pointLength;
            }
            else
            {
                groups.Add(new RegisterReadGroup
                {
                    StartAddress = groupStartAddr,
                    GroupLength = expectedNextAddr - groupStartAddr,
                    RegisterType = groupType,
                    DataPoints = groupPoints.ToList()
                });

                groupStartAddr = point.RegisterAddress;
                groupPoints = new List<DataPointConfig> { point };
                groupType = point.RegisterType;
                expectedNextAddr = groupStartAddr + pointLength;
            }
        }

        groups.Add(new RegisterReadGroup
        {
            StartAddress = groupStartAddr,
            GroupLength = expectedNextAddr - groupStartAddr,
            RegisterType = groupType,
            DataPoints = groupPoints.ToList()
        });

        return groups;
    }

    private static int GetRegisterLength(DataPointConfig point)
        => point.DataType == DataPointType.Int32 || point.DataType == DataPointType.Float ? 2 : 1;
}
```

>   分组逻辑  ：按寄存器地址排序后，检查地址是否连续（当前地址 = 上一地址 + 上一测点寄存器长度）。连续则合并为一组，不连续则另起一组。这样可以把多次请求合并为一次，减少串口通信次数。

### 步骤十三：ModbusDataCache（数据缓存）

```csharp
public class ModbusDataCache
{
    private readonly ConcurrentDictionary<int, ConcurrentDictionary<int, double>> _cache = new();

    public void UpdateDataPoint(int deviceId, int dataPointId, double value)
    {
        var deviceCache = _cache.GetOrAdd(deviceId, _ => new ConcurrentDictionary<int, double>());
        deviceCache[dataPointId] = value;
    }

    public IReadOnlyDictionary<int, double>? GetDeviceData(int deviceId)
        => _cache.TryGetValue(deviceId, out var data) ? data : null;

    public Dictionary<int, Dictionary<int, double>> GetAllData()
    {
        return _cache.ToDictionary(
            kv => kv.Key,
            kv => kv.Value.ToDictionary(inner => inner.Key, inner => inner.Value)
        );
    }

    public void Clear() => _cache.Clear();
}
```

>   为什么用     `ConcurrentDictionary`？   后台轮询线程写入，HTTP 线程读取，多线程并发访问普通 `Dictionary` 会崩溃。`ConcurrentDictionary` 内部用细粒度锁保证线程安全。

### 步骤十四：ModbusService（后台服务）

```csharp
public class ModbusService : BackgroundService, IModbusService
{
    private readonly ILogger<ModbusService> _logger;
    private readonly IConfiguration _config;
    private readonly ModbusConfigLoader _configLoader;
    private readonly ModbusPortManager _portManager;
    private readonly ModbusDataReader _dataReader;
    private readonly ModbusDataCache _cache;
    private List<ModbusDeviceConfig>? _deviceConfigs;

    public ModbusService(ILogger<ModbusService> logger, IConfiguration config,
        ModbusConfigLoader configLoader, ModbusPortManager portManager,
        ModbusDataReader dataReader, ModbusDataCache cache)
    {
        _logger = logger; _config = config; _configLoader = configLoader;
        _portManager = portManager; _dataReader = dataReader; _cache = cache;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Modbus 采集服务启动中...");

        _deviceConfigs = await _configLoader.LoadConfigsAsync();
        if (_deviceConfigs == null || _deviceConfigs.Count == 0)
        {
            _logger.LogWarning("未加载到任何设备配置，服务将空转");
            while (!stoppingToken.IsCancellationRequested)
                await Task.Delay(5000, stoppingToken);
            return;
        }

        var slaveCOMs = _deviceConfigs.Select(d => d.SlaveCOM);
        _portManager.InitializePorts(slaveCOMs);

        while (!stoppingToken.IsCancellationRequested)
        {
            try { await PollAllDevicesAsync(stoppingToken); }
            catch (OperationCanceledException) { break; }
            catch (Exception ex) { _logger.LogError(ex, "Modbus 轮询异常"); }
            await Task.Delay(1000, stoppingToken);
        }

        _logger.LogInformation("Modbus 轮询循环已退出");
    }

    private async Task PollAllDevicesAsync(CancellationToken token)
    {
        if (_deviceConfigs == null || _deviceConfigs.Count == 0) return;

        var devicesByCOM = _deviceConfigs
            .GroupBy(d => GetMasterCOM(d.SlaveCOM))
            .Where(g => g.Key != null)
            .ToDictionary(g => g.Key!, g => g.ToList());

        var portTasks = devicesByCOM.Select(async kvp =>
        {
            var (masterCOM, deviceConfigs) = (kvp.Key, kvp.Value);
            var port = _portManager.GetPort(masterCOM);
            if (port == null)
            {
                _logger.LogWarning("主站端口 {COM} 不可用", masterCOM);
                return;
            }
            var master = _dataReader.GetOrCreateMaster(port);
            foreach (var deviceConfig in deviceConfigs)
            {
                if (token.IsCancellationRequested) break;
                await _dataReader.PollDeviceAsync(master, deviceConfig, token);
            }
        });

        await Task.WhenAll(portTasks);
    }

    private string? GetMasterCOM(string slaveCOM)
    {
        var mappings = _config.GetSection("Modbus:PortMappings")
            .Get<Dictionary<string, string>>() ?? new Dictionary<string, string>();
        var slaveToMaster = mappings.ToDictionary(m => m.Value, m => m.Key);
        return slaveToMaster.TryGetValue(slaveCOM, out var masterCOM) ? masterCOM : null;
    }

    public override async Task StopAsync(CancellationToken cancellationToken)
    {
        _logger.LogInformation("Modbus 采集服务正在停止...");
        _portManager.Dispose();
        _dataReader.ClearMasterCache();
        await base.StopAsync(cancellationToken);
        _logger.LogInformation("Modbus 采集服务已停止");
    }

    public Dictionary<int, Dictionary<int, double>> GetCurrentData()
        => _cache.GetAllData();

    public async Task ReloadConfigsAsync()
    {
        _logger.LogInformation("重新加载设备配置...");
        _portManager.Dispose();
        _dataReader.ClearMasterCache();
        _deviceConfigs = await _configLoader.LoadConfigsAsync();
        if (_deviceConfigs != null && _deviceConfigs.Count > 0)
        {
            var slaveCOMs = _deviceConfigs.Select(d => d.SlaveCOM);
            _portManager.InitializePorts(slaveCOMs);
        }
    }
}
```

>   BackgroundService  ：继承 `BackgroundService` 并重写 `ExecuteAsync`，框架会自动在应用启动时运行、退出时传递 `CancellationToken`。`AddHostedService<ModbusService>()` 将其注册为单例后台服务。

| -------------- | -------------------------------------- | ---------------- | ------------------------- |

-   Repository 层  ：操作 Entity，返回 Entity
-   Service 层  ：通过 Repository 操作 Entity，进行业务校验，，，不返回 Entity，，（返回 DTO）

### 任务三：HttpClient 封装 + Token 管理

#### 3a. TokenStore.cs — 内存中保存 Token（单例，线程安全）

```csharp
namespace Client.Infrastructure
{
    public class TokenStore
    {
        private string? _accessToken;
        private string? _refreshToken;
        private readonly object _lock = new();

        public string? AccessToken
        {
            get { lock (_lock) { return _accessToken; } }
            set { lock (_lock) { _accessToken = value; } }
        }

        public string? RefreshToken
        {
            get { lock (_lock) return _refreshToken; }
            set { lock (_lock) _refreshToken = value; }
        }

        public void SetTokens(string accessToken, string refreshToken)
        {
            lock (_lock)
            {
                _accessToken = accessToken;
                _refreshToken = refreshToken;
            }
        }

        public void ClearTokens()
        {
            lock (_lock)
            {
                _accessToken = null;
                _refreshToken = null;
            }
        }
    }
}
```

>   为什么要注册为单例？   Token 在整个应用生命周期中共享。如果注册为 Transient，每次注入都创建新实例，Token 会被丢失。使用 `lock` 保证多线程访问安全。

#### 3b. HttpClientBase.cs — HTTP 基类，自动附加 Token + 401 自动刷新

```csharp
using Client.Models.Dtos;
using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Text.Json;

namespace Client.Infrastructure
{
    public class HttpClientBase
    {
        private readonly HttpClient _httpClient;
        private readonly TokenStore _tokenStore;
        private readonly SemaphoreSlim _refreshTokenLock = new(1, 1);
        private static readonly JsonSerializerOptions _jsonOptions = new()
        {
            PropertyNameCaseInsensitive = true
        };

        private const string BaseUrl = "http://localhost:5078/";

        public HttpClientBase(TokenStore tokenStore)
        {
            _httpClient = new HttpClient { BaseAddress = new Uri(BaseUrl) };
            _tokenStore = tokenStore;
        }

        // ====== 公开方法 ======
        public async Task<HttpResponseMessage> GetAsync(string url)
            => await SendAsync(() => new HttpRequestMessage(HttpMethod.Get, url));

        public async Task<HttpResponseMessage> PostAsync<T>(string url, T body)
        {
            var request = new HttpRequestMessage(HttpMethod.Post, url)
            {
                Content = JsonContent.Create(body, options: _jsonOptions)
            };
            return await SendAsync(() => request);
        }

        public async Task<HttpResponseMessage> PutAsync<T>(string url, T body)
        {
            var request = new HttpRequestMessage(HttpMethod.Put, url)
            {
                Content = JsonContent.Create(body, options: _jsonOptions)
            };
            return await SendAsync(() => request);
        }

        public async Task<HttpResponseMessage> DeleteAsync(string url)
            => await SendAsync(() => new HttpRequestMessage(HttpMethod.Delete, url));

        // ====== 核心发送逻辑 ======
        private async Task<HttpResponseMessage> SendAsync(Func<HttpRequestMessage> requestFactory)
        {
            var request = requestFactory();
            AttachToken(request);
            var response = await _httpClient.SendAsync(request);

            if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
            {
                var refreshed = await TryRefreshTokenAsync();
                if (refreshed)
                {
                    request = requestFactory();
                    AttachToken(request);
                    response = await _httpClient.SendAsync(request);
                }
            }

            return response;
        }

        private void AttachToken(HttpRequestMessage request)
        {
            var token = _tokenStore.AccessToken;
            if (!string.IsNullOrWhiteSpace(token))
            {
                request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
            }
        }

        private async Task<bool> TryRefreshTokenAsync()
        {
            await _refreshTokenLock.WaitAsync();
            try
            {
                var response = await _httpClient.PostAsJsonAsync("api/auth/refresh", new TokenStore
                {
                    AccessToken = _tokenStore.AccessToken,
                    RefreshToken = _tokenStore.RefreshToken
                }, _jsonOptions);

                if (!response.IsSuccessStatusCode) return false;

                var result = await response.Content.ReadFromJsonAsync<ApiResult<LoginResponse>>(_jsonOptions);
                if (result == null || result.Data == null || result.Code != 200) return false;

                _tokenStore.SetTokens(result.Data.AccseeToken, result.Data.RefreshToken);
                return true;
            }
            catch (Exception)
            {
                return false;
            }
            finally
            {
                _refreshTokenLock.Release();
            }
        }
    }
}
```

>   关键设计  ：
>
> -   BaseUrl 末尾带     `/`  ：`http://localhost:5078/`。HttpClient 拼接相对路径时，BaseAddress 必须以 `/` 结尾，否则相对路径会替换整个路径段。
> -   相对路径不带前导     `/`  ：如 `api/devices`（不是 `/api/devices`），确保与 BaseAddress 正确拼接。
> -   SemaphoreSlim(1,1)  ：异步锁，同一时刻只允许一个线程进入刷新逻辑，防止并发刷新。
> -   PropertyNameCaseInsensitive = true  ：JSON 反序列化时忽略属性名大小写，匹配服务端返回的 camelCase。
> -   requestFactory 委托  ：重放请求时需要重新创建 HttpRequestMessage（它的 Content 是一次性流），不能用同一个实例。

#### 5b. LoginViewModel.cs — 登录逻辑

```csharp
using Client.Service;
using System.Net.Http;

namespace Client.ViewModels
{
    public class LoginViewModel : BindableBase
    {
        private readonly IAuthService _authService;
        private readonly IShellService _shellService;

        private string _username = string.Empty;
        private bool _hasError;
        private string? _errorMessage;

        public AsyncDelegateCommand<object> LoginCommand { get; }

        public LoginViewModel(IAuthService authService, IShellService shellService)
        {
            _authService = authService;
            _shellService = shellService;
            LoginCommand = new AsyncDelegateCommand<object>(ExecuteLogin);
        }

        public string Username
        {
            get => _username;
            set => SetProperty(ref _username, value);
        }

        public bool HasError
        {
            get => _hasError;
            set => SetProperty(ref _hasError, value);
        }

        public string? ErrorMessage
        {
            get => _errorMessage;
            set => SetProperty(ref _errorMessage, value);
        }

        private async Task ExecuteLogin(object parameter)
        {
            if (!(parameter is PasswordBox passwordBox)) return;

            string password = passwordBox.Password;

            if (string.IsNullOrWhiteSpace(password) || string.IsNullOrWhiteSpace(Username))
            {
                await ShowErrorMessageAsync("账号或密码不能为空");
                return;
            }

            try
            {
                var success = await _authService.SignInAsync(Username, password);
                if (success)
                {
                    _shellService.LoginToShell();
                }
                else
                {
                    await ShowErrorMessageAsync("账号或密码错误");
                }
            }
            catch (HttpRequestException)
            {
                await ShowErrorMessageAsync("无法连接服务器，请检查网络");
            }
            catch (Exception)
            {
                await ShowErrorMessageAsync("登录失败，请稍后重试");
            }
        }

        private async Task ShowErrorMessageAsync(string message)
        {
            HasError = true;
            ErrorMessage = message;
            await Task.Delay(2000);
            HasError = false;
            ErrorMessage = string.Empty;
        }
    }
}
```

>   为什么用 AsyncDelegateCommand 而不是 DelegateCommand？   登录涉及网络请求（async），DelegateCommand 不支持 async void 中的异常安全捕获。AsyncDelegateCommand 返回 Task，异常不会逃逸。
>
>   ShellService 解耦  ：登录成功后调用 `_shellService.LoginToShell()`，而不是在 VM 中直接创建 ShellView。这样 LoginViewModel 不需要知道 ShellView 的存在，也避免了 RegionManager 绑定问题。

```csharp
using Client.Models.Dtos;

namespace Client.Modules.DeviceManagement.ViewModels
{

        private bool CanNext() => pageIndex < maxPage;

        {
            if (CanPrev()) { pageIndex--; await LoadData(); }

        {
            if (CanNext()) { pageIndex++; await LoadData(); }
        }

        {
            var devicePage = await _deviceService.GetPagedAsync(pageIndex, pageSize);
            if (devicePage != null)

                Devices = devicePage.Items;
                totalCount = devicePage.TotalCount;

                RaisePropertyChanged(nameof(maxPage));

                NextPageCommand.RaiseCanExecuteChanged();

            _dialogService.ShowDialog("DeviceEditView", parameters, async result =>
            {
                if (result.Result == ButtonResult.OK)
                    await LoadData();
            });
        }

            if (id == null) return;
            var device = await _deviceService.GetByIdAsync((int)id);
            if (device != null)
            {
                var parameters = new DialogParameters();
                parameters.Add("device", device);
                _dialogService.ShowDialog(nameof(DeviceEditView), parameters, async result =>

```

>   EditCommand 使用     `int?`     而非     `int`  ：DataGrid 行中的按钮通过 `CommandParameter="{Binding Id}"` 传递 Id。但在行未完全加载或 DataGrid 模板初始化阶段，Id 可能为 `DependencyProperty.UnsetValue`（非 null，但也不是 int），强类型转换会抛 NullReferenceException。使用 `int?` 可以安全接收并提前 return。

#### 6d. DeviceEditViewModel.cs — 编辑表单逻辑（Prism 9.0 IDialogAware）

```csharp
using Client.Infrastructure;
using Client.Models.Dtos;
using Client.Service;

namespace Client.Modules.DeviceManagement.ViewModels
{
    public class DeviceEditViewModel : BindableBase, IDialogAware
    {
        private readonly IDeviceService _deviceService;
        private readonly IProductionLineService _productionLineService;

        private string _name = string.Empty;
        private string _code = string.Empty;
        private List<ProductionLineDto>? _productionLines;
        private int _productionLineId;
        private byte _slaveAddress;
        private string _status = string.Empty;
        private string _comPort = "COM2";
        private string? _description;
        private string _errorMessage = string.Empty;
        private bool _hasError;
        private int _editId;

        public AsyncDelegateCommand SaveCommand { get; }
        public DelegateCommand CancelCommand { get; }

        public DeviceEditViewModel(IDeviceService deviceService, IProductionLineService productionLineService)
        {
            _deviceService = deviceService;
            _productionLineService = productionLineService;

            CancelCommand = new DelegateCommand(ExecuteCancel);
            SaveCommand = new AsyncDelegateCommand(ExecuteSave);
        }

        // ====== IDialogAware 实现 ========
        public DialogCloseListener RequestClose { get; }

        public bool CanCloseDialog() => true;
        public void OnDialogClosed() { }

        public void OnDialogOpened(IDialogParameters parameters)
        {
            _ = LoadAsync(parameters);  // Fire-and-Forget 模式
        }

        // 异步加载：产线列表 + 编辑模式回填
        private async Task LoadAsync(IDialogParameters parameters)
        {
            try
            {
                ProductionLines = await _productionLineService.GetAllAsync();

                if (parameters.TryGetValue("device", out DeviceDto device) && device != null)
                {
                    _editId = device.Id;
                    Name = device.Name;
                    Code = device.Code;
                    ProductionLineId = device.ProductionLineId;
                    SlaveAddress = device.SlaveAddress;
                    ComPort = device.ComPort;
                    Description = device.Description;
                }
            }
            catch (Exception ex)
            {
                ErrorMessage = $"加载数据失败：{ex.Message}";
                HasError = true;
            }
        }

        private async Task ExecuteSave()
        {
            _hasError = false;
            if (string.IsNullOrWhiteSpace(Name) || string.IsNullOrWhiteSpace(Code))
            {
                ErrorMessage = "设备名称和编码不能为空";
                HasError = true;
                return;
            }

            bool success;
            if (_editId > 0)
            {
                success = await _deviceService.UpdateAsync(_editId, new UpdateDeviceRequest
                {
                    Name = Name, Code = Code, ProductionLineId = ProductionLineId,
                    SlaveAddress = SlaveAddress, ComPort = ComPort, Description = Description
                });
            }
            else
            {
                success = await _deviceService.CreateAsync(new CreateDeviceRequest
                {
                    Name = Name, Code = Code, ProductionLineId = ProductionLineId,
                    SlaveAddress = SlaveAddress, ComPort = ComPort, Description = Description
                });
            }

            if (success)
            {
                RequestClose.Invoke(new DialogResult(ButtonResult.OK));
            }
            else
            {
                _hasError = true;
                ErrorMessage = "保存失败，请检查数据或网络连接";
            }
        }

        private void ExecuteCancel()
        {
            RequestClose.Invoke(new DialogResult(ButtonResult.Cancel));
        }

        #region 绑定属性
        public string Name { get => _name; set => SetProperty(ref _name, value); }
        public string Code { get => _code; set => SetProperty(ref _code, value); }
        public List<ProductionLineDto>? ProductionLines
        {
            get => _productionLines;
            set => SetProperty(ref _productionLines, value);
        }
        public int ProductionLineId { get => _productionLineId; set => SetProperty(ref _productionLineId, value); }
        public byte SlaveAddress { get => _slaveAddress; set => SetProperty(ref _slaveAddress, value); }
        public string ComPort { get => _comPort; set => SetProperty(ref _comPort, value); }
        public string Status { get => _status; set => SetProperty(ref _status, value); }
        public string? Description { get => _description; set => SetProperty(ref _description, value); }
        public bool HasError { get => _hasError; set => SetProperty(ref _hasError, value); }
        public string ErrorMessage
        {
            get => _errorMessage;
            set => SetProperty(ref _errorMessage, value);
        }
        #endregion
    }
}
```

>   OnDialogOpened 的异步模式  ：`IDialogAware.OnDialogOpened` 返回 `void`，不能声明为 `async`。使用 Fire-and-Forget 模式 `_ = LoadAsync(parameters)`，将异步操作委托给返回 Task 的方法。如果使用 `async void`，内部异常会逃逸到同步上下文导致崩溃；Fire-and-Forget 模式下由 LoadAsync 内部 try-catch 捕获。
>
>   Prism 9.0 IDialogAware 变化  ：
>
> - `RequestClose` 从 `event Action<IDialogResult>` 变为 `DialogCloseListener` 属性
> - 关闭弹窗调用 `RequestClose.Invoke(new DialogResult(ButtonResult.OK))`
> - 命名空间从 `Prism.Services.Dialogs` 变为 `Prism.Dialogs`

### 任务七：DataGrid 自定义样式

为了让 DataGrid 显示美观、风格统一，创建了独立的样式资源字典 `Styles/DataGridStyle.xaml`，在 App.xaml 中全局引用。

样式要点：

-   表头  ：浅灰背景（#F5F7FA）、居中对齐、右侧细边框分隔、排序箭头
-   行  ：透明背景 + 悬停变浅灰（#FAFAFA）+ 交替行（#F9F9F9）+ 底部细分隔线
-   单元格  ：居中对齐、右侧细边框（#E8E8E8）、选中无效果（覆盖系统选中画刷为透明）
-   无选中效果  ：覆盖 `SystemColors.HighlightBrushKey` 等系统画刷，确保选中行文字仍为黑色

```xml
<!-- 样式关键片段 -->
<Style x:Key="MinimalDataGridStyle" TargetType="DataGrid">

                                    BorderBrush="#E8E8E8" BorderThickness="0,0,0,1">

                                </SelectiveScrollingGrid>

                <Setter Property="VerticalContentAlignment" Value="Center"/>

        </Setter.Value>
    </Setter>
</Style>
```

>   选中无效果的原因  ：WPF 默认选中行背景变蓝、文字变白。通过自定义 CellTemplate（不含系统选中触发器）和覆盖系统画刷，确保选中时背景透明、文字仍为黑色。如果只在行级别覆盖而未在单元格级别覆盖，系统默认的单元格选中样式仍会覆盖行样式。
